ADD-ON: LOCK SCREEN FLOATING CONTROL DIALOG FOR PDF WALLPAPER

FEATURE OVERVIEW

When user sets a PDF as wallpaper, a premium floating control dialog appears on the lock screen that allows manual page navigation WITHOUT unlocking the phone.

---

3. IMPLEMENTATION CODE

A. Floating Dialog Service

```kotlin
class PDFLockScreenDialogService : Service() {
    
    private lateinit var windowManager: WindowManager
    private lateinit var floatingView: View
    private var currentPage = 0
    private var totalPages = 0
    private var isAutoRotate = true
    private var isDialogVisible = true
    
    companion object {
        const val ACTION_NEXT = "NEXT_PAGE"
        const val ACTION_PREV = "PREV_PAGE"
        const val ACTION_PAUSE = "PAUSE_AUTO"
        const val ACTION_CLOSE = "CLOSE_DIALOG"
        const val EXTRA_PAGE = "current_page"
        const val EXTRA_TOTAL = "total_pages"
    }
    
    override fun onCreate() {
        super.onCreate()
        windowManager = getSystemService(WINDOW_SERVICE) as WindowManager
        createFloatingDialog()
        
        // Register broadcast receiver for control actions
        registerReceiver(controlReceiver, IntentFilter().apply {
            addAction(ACTION_NEXT)
            addAction(ACTION_PREV)
            addAction(ACTION_PAUSE)
            addAction(ACTION_CLOSE)
        })
    }
    
    private fun createFloatingDialog() {
        // Inflate the floating dialog layout
        floatingView = LayoutInflater.from(this).inflate(
            R.layout.dialog_lock_screen_pdf_controls, 
            null
        )
        
        // Setup dialog parameters
        val params = WindowManager.LayoutParams(
            WindowManager.LayoutParams.MATCH_PARENT,
            WindowManager.LayoutParams.WRAP_CONTENT,
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O)
                WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY
            else
                WindowManager.LayoutParams.TYPE_PHONE,
            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or
            WindowManager.LayoutParams.FLAG_WATCH_OUTSIDE_TOUCH or
            WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN,
            PixelFormat.TRANSLUCENT
        ).apply {
            gravity = Gravity.BOTTOM or Gravity.CENTER_HORIZONTAL
            y = 100 // 100dp from bottom
            width = (resources.displayMetrics.widthPixels * 0.9).toInt() // 90% of screen width
        }
        
        // Add view to window
        windowManager.addView(floatingView, params)
        
        // Setup UI elements
        setupUI()
        
        // Animate show
        animateShow()
    }
    
    private fun setupUI() {
        // Page indicator
        val tvPageNumber = floatingView.findViewById<TextView>(R.id.tv_page_number)
        val tvTotalPages = floatingView.findViewById<TextView>(R.id.tv_total_pages)
        val progressBar = floatingView.findViewById<ProgressBar>(R.id.progress_bar)
        val tvPageTitle = floatingView.findViewById<TextView>(R.id.tv_page_title)
        
        // Buttons
        val btnPrevious = floatingView.findViewById<ImageButton>(R.id.btn_previous)
        val btnNext = floatingView.findViewById<ImageButton>(R.id.btn_next)
        val btnPause = floatingView.findViewById<ImageButton>(R.id.btn_pause)
        val btnSettings = floatingView.findViewById<ImageButton>(R.id.btn_settings)
        val btnClose = floatingView.findViewById<ImageButton>(R.id.btn_close)
        
        // Update UI with current data
        updateUI(currentPage, totalPages)
        
        // Set click listeners
        btnPrevious.setOnClickListener {
            if (currentPage > 0) {
                currentPage--
                updateWallpaper(currentPage)
                updateUI(currentPage, totalPages)
                showPageIndicator("Page ${currentPage + 1}/$totalPages")
                vibrate()
            }
        }
        
        btnNext.setOnClickListener {
            if (currentPage < totalPages - 1) {
                currentPage++
                updateWallpaper(currentPage)
                updateUI(currentPage, totalPages)
                showPageIndicator("Page ${currentPage + 1}/$totalPages")
                vibrate()
            }
        }
        
        btnPause.setOnClickListener {
            isAutoRotate = !isAutoRotate
            btnPause.setImageResource(
                if (isAutoRotate) R.drawable.ic_pause 
                else R.drawable.ic_play
            )
            updateWallpaperService(isAutoRotate)
            showPageIndicator(if (isAutoRotate) "Auto-Rotate: ON" else "Auto-Rotate: PAUSED")
        }
        
        btnSettings.setOnClickListener {
            // Open PDF settings dialog
            showPDFSettingsDialog()
        }
        
        btnClose.setOnClickListener {
            hideDialog()
        }
        
        // Make dialog draggable
        makeDraggable(floatingView)
    }
    
    private fun updateUI(page: Int, total: Int) {
        val tvPageNumber = floatingView.findViewById<TextView>(R.id.tv_page_number)
        val tvTotalPages = floatingView.findViewById<TextView>(R.id.tv_total_pages)
        val progressBar = floatingView.findViewById<ProgressBar>(R.id.progress_bar)
        
        tvPageNumber.text = "${page + 1}"
        tvTotalPages.text = "of $total"
        progressBar.max = total
        progressBar.progress = page + 1
        
        // Update page title if available
        val tvPageTitle = floatingView.findViewById<TextView>(R.id.tv_page_title)
        // Fetch page title from PDF metadata
    }
    
    private fun updateWallpaper(page: Int) {
        // Update the actual lock screen wallpaper to new page
        val pdfPath = getPDFPath()
        setPDFWallpaper(pdfPath, page)
        
        // Broadcast to update notification
        sendBroadcast(Intent("UPDATE_WALLPAPER").apply {
            putExtra(EXTRA_PAGE, page)
        })
    }
    
    private fun showPageIndicator(message: String) {
        // Create temporary overlay showing page number
        val overlay = LayoutInflater.from(this).inflate(
            R.layout.overlay_page_indicator, 
            null
        )
        overlay.findViewById<TextView>(R.id.tv_indicator).text = message
        
        val params = WindowManager.LayoutParams(
            WindowManager.LayoutParams.WRAP_CONTENT,
            WindowManager.LayoutParams.WRAP_CONTENT,
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O)
                WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY
            else
                WindowManager.LayoutParams.TYPE_PHONE,
            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE,
            PixelFormat.TRANSLUCENT
        ).apply {
            gravity = Gravity.CENTER
        }
        
        windowManager.addView(overlay, params)
        
        // Remove after 2 seconds
        Handler(Looper.getMainLooper()).postDelayed({
            try {
                windowManager.removeView(overlay)
            } catch (e: Exception) {
                // View already removed
            }
        }, 2000)
    }
    
    private fun makeDraggable(view: View) {
        var initialX = 0
        var initialY = 0
        var initialTouchX = 0f
        var initialTouchY = 0f
        
        view.setOnTouchListener { _, event ->
            when (event.action) {
                MotionEvent.ACTION_DOWN -> {
                    initialX = (view.layoutParams as WindowManager.LayoutParams).x
                    initialY = (view.layoutParams as WindowManager.LayoutParams).y
                    initialTouchX = event.rawX
                    initialTouchY = event.rawY
                    true
                }
                MotionEvent.ACTION_MOVE -> {
                    val params = view.layoutParams as WindowManager.LayoutParams
                    params.x = initialX + (event.rawX - initialTouchX).toInt()
                    params.y = initialY + (event.rawY - initialTouchY).toInt()
                    windowManager.updateViewLayout(view, params)
                    true
                }
                else -> false
            }
        }
    }
    
    private fun animateShow() {
        floatingView.apply {
            translationY = 200f
            alpha = 0f
            animate()
                .translationY(0f)
                .alpha(1f)
                .setDuration(500)
                .setInterpolator(DecelerateInterpolator())
                .start()
        }
    }
    
    private fun animateHide() {
        floatingView.apply {
            animate()
                .translationY(200f)
                .alpha(0f)
                .setDuration(300)
                .setInterpolator(AccelerateInterpolator())
                .withEndAction {
                    windowManager.removeView(floatingView)
                }
                .start()
        }
    }
    
    private fun hideDialog() {
        isDialogVisible = false
        animateHide()
        // Show notification to reopen
        showReopenNotification()
    }
    
    private fun showReopenNotification() {
        // Notification to bring back the dialog
        val intent = Intent(this, PDFLockScreenDialogService::class.java)
        intent.action = "SHOW_DIALOG"
        val pendingIntent = PendingIntent.getService(
            this, 0, intent, 
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )
        
        val notification = NotificationCompat.Builder(this, "pdf_control_channel")
            .setContentTitle("📄 PDF Controls Hidden")
            .setContentText("Tap to show controls on lock screen")
            .setSmallIcon(R.drawable.ic_pdf)
            .setContentIntent(pendingIntent)
            .setVisibility(NotificationCompat.VISIBILITY_PUBLIC)
            .build()
        
        val manager = getSystemService(NotificationManager::class.java)
        manager.notify(1001, notification)
    }
    
    private fun vibrate() {
        val vibrator = getSystemService(VIBRATOR_SERVICE) as Vibrator
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            vibrator.vibrate(VibrationEffect.createOneShot(50, VibrationEffect.DEFAULT_AMPLITUDE))
        } else {
            vibrator.vibrate(50)
        }
    }
    
    private val controlReceiver = object : BroadcastReceiver() {
        override fun onReceive(context: Context, intent: Intent) {
            when (intent.action) {
                "SHOW_DIALOG" -> {
                    if (!isDialogVisible) {
                        isDialogVisible = true
                        createFloatingDialog()
                    }
                }
                "UPDATE_PAGE" -> {
                    currentPage = intent.getIntExtra(EXTRA_PAGE, 0)
                    totalPages = intent.getIntExtra(EXTRA_TOTAL, 20)
                    updateUI(currentPage, totalPages)
                }
            }
        }
    }
    
    override fun onDestroy() {
        super.onDestroy()
        try {
            windowManager.removeView(floatingView)
        } catch (e: Exception) {
            // View already removed
        }
        unregisterReceiver(controlReceiver)
    }
}
```

5. HOW TO START THE SERVICE

When PDF is set as wallpaper:

```kotlin
private fun setPDFAsWallpaper(pdfPath: String, totalPages: Int) {
    // Set wallpaper logic...
    
    // Start lock screen dialog service
    val intent = Intent(this, PDFLockScreenDialogService::class.java)
    intent.putExtra("PDF_PATH", pdfPath)
    intent.putExtra("TOTAL_PAGES", totalPages)
    intent.putExtra("START_PAGE", 0)
    
    if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
        startForegroundService(intent)
    } else {
        startService(intent)
    }
}
```
